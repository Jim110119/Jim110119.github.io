-- 核心 RPC 存储过程：卡密核销（只更新这个函数，绝不碰表结构）
CREATE OR REPLACE FUNCTION verify_code(p_username TEXT, p_code TEXT)
RETURNS json AS $$
DECLARE
    v_is_used BOOLEAN;
    v_code_exists BOOLEAN;
BEGIN
    -- 第一关：检查卡密到底存不存在？
    SELECT EXISTS(SELECT 1 FROM activation_codes WHERE code = p_code) INTO v_code_exists;
    IF NOT v_code_exists THEN
        RETURN json_build_object('success', false, 'msg', '无效的卡密，请检查是否输入错误');
    END IF;

    -- 第二关：检查卡密是不是被人用过了？
    SELECT is_used INTO v_is_used FROM activation_codes WHERE code = p_code;
    IF v_is_used THEN
        -- 👇 你的专属文案已经替换在这里！
        RETURN json_build_object('success', false, 'msg', '该卡密已被使用，请重新购买或联系客服');
    END IF;

    -- 第三关：两表连动，瞬间核销！(事务级别的原子操作，绝对安全)
    -- a. 作废卡密，记录元凶
    UPDATE activation_codes 
    SET is_used = true, 
        used_by_user = p_username, 
        used_time = NOW() 
    WHERE code = p_code;

    -- b. 给用户发放永久 VIP 特权
    UPDATE app_users 
    SET is_vip = true 
    WHERE username = p_username;

    -- 大功告成，返回成功信号
    RETURN json_build_object('success', true, 'msg', '激活成功！高级功能已永久解锁');
END;
$$ LANGUAGE plpgsql;


-- ==========================================
-- 1. 注册新账号 (V2 升维版)
-- ==========================================
CREATE OR REPLACE FUNCTION register_account_v2(p_username TEXT, p_password TEXT, p_uuid TEXT)
RETURNS json AS $$
BEGIN
    -- 1. 查重：看看用户名是不是被抢占了
    IF EXISTS (SELECT 1 FROM app_users WHERE username = p_username) THEN
        RETURN json_build_object('success', false, 'msg', '该账号已被注册，请换一个名字');
    END IF;

    -- 2. 放行：写入数据库，默认不是 VIP
    INSERT INTO app_users (username, password, device_uuid, is_vip)
    VALUES (p_username, p_password, p_uuid, false);

    RETURN json_build_object('success', true, 'msg', '注册成功');
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;


-- ==========================================
-- 2. 账号密码登录验证 (V2 升维版)
-- ==========================================
CREATE OR REPLACE FUNCTION login_device_v2(p_username TEXT, p_password TEXT, p_uuid TEXT)
RETURNS json AS $$
DECLARE
    v_db_password TEXT;
    v_db_uuid TEXT;
    v_is_vip BOOLEAN;
BEGIN
    -- 1. 查账号是否存在
    SELECT password, device_uuid, is_vip 
    INTO v_db_password, v_db_uuid, v_is_vip 
    FROM app_users 
    WHERE username = p_username;

    IF NOT FOUND THEN
        RETURN json_build_object('success', false, 'msg', '账号不存在，请先注册');
    END IF;

    -- 2. 校验密码
    IF v_db_password != p_password THEN
        RETURN json_build_object('success', false, 'msg', '密码错误，请重新输入');
    END IF;

    -- 3. 智能设备锁校验
    -- 如果数据库里是 NULL 或者空字符串，说明是新注册或刚解绑，直接绑上新设备！
    IF (v_db_uuid IS NULL OR v_db_uuid = '') THEN
        UPDATE app_users SET device_uuid = p_uuid WHERE username = p_username;
        RETURN json_build_object('success', true, 'is_vip', v_is_vip, 'msg', '登录成功，已绑定当前车机');
        
    -- 如果数据库里有值，且跟当前来敲门的设备码不一样，无情拦截！
    ELSIF v_db_uuid != p_uuid THEN
        RETURN json_build_object('success', false, 'msg', '该账号已绑定其他车机！如需换绑，请在原车机退出或联系客服');
    END IF;

    -- 4. 正常情况：已绑定，且设备码一致，直接放行
    RETURN json_build_object('success', true, 'is_vip', v_is_vip, 'msg', '登录成功，欢迎回来');
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;


-- ==========================================
-- 3. 安全退出解绑 (V2 升维版)
-- ==========================================
CREATE OR REPLACE FUNCTION unbind_device_v2(p_username TEXT, p_uuid TEXT)
RETURNS json AS $$
DECLARE
    v_real_uuid TEXT;
BEGIN
    -- 1. 尝试查找该账号
    SELECT device_uuid INTO v_real_uuid FROM app_users WHERE username = p_username;

    -- 2. 如果因为某种原因没找到（防错机制）
    IF NOT FOUND THEN
        RETURN json_build_object('success', false, 'msg', '账号不存在');
    END IF;

    -- 3. 安全校验：防止别人拿着你的用户名恶意解绑
    IF v_real_uuid != p_uuid AND v_real_uuid IS NOT NULL AND v_real_uuid != '' THEN
        RETURN json_build_object('success', false, 'msg', '当前车机与校验绑定的设备不符，拒绝解绑');
    END IF;

    -- 4. 执行解绑：把该账号绑定的车机特征码清空
    UPDATE app_users SET device_uuid = '' WHERE username = p_username;

    RETURN json_build_object('success', true, 'msg', '解绑成功，您可以随时在其他车机登录该账号');
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ==========================================
-- 5. 终端状态暗哨审计 (V3 蜜罐探针)
-- 作用：让终端主动上报它的 VIP 状态，服务器进行测谎
-- ==========================================
CREATE OR REPLACE FUNCTION audit_client_status(p_username TEXT, p_client_is_vip BOOLEAN)
RETURNS json AS $$
DECLARE
    v_db_is_vip BOOLEAN;
BEGIN
    -- 1. 查出服务器里记录的绝对真相
    SELECT is_vip INTO v_db_is_vip FROM app_users WHERE username = p_username;

    IF NOT FOUND THEN
        RETURN json_build_object('success', false, 'msg', '账号不存在');
    END IF;

    -- 2. 🌟 核心测谎逻辑：信息差比对 🌟
    -- 如果服务器里他明明不是 VIP，但他上传的参数却说自己是 VIP...
    IF v_db_is_vip = false AND p_client_is_vip = true THEN
        
        -- 抓到内鬼了！悄悄给他打上破解标签，记录作案时间
        UPDATE app_users 
        SET is_cracked = true, 
            crack_report_time = NOW() 
        WHERE username = p_username;
        
        -- 注意：这里依然返回 success，不打草惊蛇！让他在终端继续玩，但你的后台已经把他拉黑了。
        RETURN json_build_object('success', true, 'msg', '状态同步正常(假象)');
    END IF;

    -- 正常情况
    RETURN json_build_object('success', true, 'msg', '状态同步正常');
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;


-- ==========================================
-- 5. 终端状态暗哨审计 (V4 支持无账号抓捕版)
-- ==========================================
CREATE OR REPLACE FUNCTION audit_client_status(p_username TEXT, p_uuid TEXT, p_client_is_vip BOOLEAN)
RETURNS json AS $$
DECLARE
    v_db_is_vip BOOLEAN := false;
BEGIN
    -- 1. 优先按用户名查（如果他登录了的话）
    IF p_username IS NOT NULL AND p_username != '' THEN
        SELECT is_vip INTO v_db_is_vip FROM app_users WHERE username = p_username;
    
    -- 2. 如果没登录，按车机设备码查
    ELSIF p_uuid IS NOT NULL AND p_uuid != '' THEN
        SELECT is_vip INTO v_db_is_vip FROM app_users WHERE device_uuid = p_uuid LIMIT 1;
    END IF;

    -- 3. 🌟 终极裁决 🌟
    -- 如果数据库里查无此人，或者查出来了但他不是 VIP...
    -- 但是他居然敢向服务器宣称自己是 VIP (p_client_is_vip = true)
    IF (v_db_is_vip IS NULL OR v_db_is_vip = false) AND p_client_is_vip = true THEN
        
        -- 这绝对是个破解者！就算他连账号都没注册，我们也要把他的设备码给拉黑拉满！
        -- 如果他有账号，就更新他的案底
        IF p_username IS NOT NULL AND p_username != '' THEN
            UPDATE app_users SET is_cracked = true, crack_report_time = NOW() WHERE username = p_username;
        ELSE
            -- 如果他连账号都没有，我们把他的设备码记录下来
            -- (前提是你的表里支持对 device_uuid 进行更新操作，或者你可以新建个 cracked_devices 专门装这些人)
            UPDATE app_users SET is_cracked = true, crack_report_time = NOW() WHERE device_uuid = p_uuid;
        END IF;
        
        RETURN json_build_object('success', true, 'msg', '状态同步正常(假象)');
    END IF;

    RETURN json_build_object('success', true, 'msg', '状态同步正常');
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

ALTER TABLE app_users

-- ==========================================
-- 步骤二：升维重写 login_device_v2 登录函数
-- （直接覆盖旧函数，不影响原有业务，无需改动 App）
-- ==========================================
CREATE OR REPLACE FUNCTION login_device_v2(p_username TEXT, p_password TEXT, p_uuid TEXT)
RETURNS json AS $$
DECLARE
    v_user RECORD;
BEGIN
    -- 1. 查账号是否存在，并把该账号所有信息一次性拉取到记录变量 v_user 中
    SELECT * INTO v_user
    FROM app_users
    WHERE username = p_username;

    IF NOT FOUND THEN
        RETURN json_build_object('success', false, 'msg', '账号不存在，请检查是否输入错误');
    END IF;

    -- 2. 核对密码
    IF v_user.password != p_password THEN
        RETURN json_build_object('success', false, 'msg', '密码错误，请重新输入');
    END IF;

    -- ==========================================
    -- 🌟 3. 核心：设备匹配与智能换绑引擎
    -- ==========================================

    -- 【情景 A】: 账号纯新没绑过设备，或者就是本机正常的重复登录
    IF v_user.device_uuid IS NULL OR v_user.device_uuid = '' OR v_user.device_uuid = p_uuid THEN
        -- 放行，顺手稳固一下设备绑定（不增加风控计数）
        UPDATE app_users SET device_uuid = p_uuid WHERE username = p_username;
        -- 返回时带上 is_vip 字段，兼容你 App 原有的逻辑
        RETURN json_build_object('success', true, 'msg', '登录成功', 'is_vip', v_user.is_vip);
    END IF;

    -- 【情景 B】: 发现设备不匹配！触发“自动换绑”防线
    IF v_user.device_uuid != p_uuid THEN
        
        -- 判断一：如果距离上次换绑已经超过 30 天了（或者从来没换绑过），给用户“恢复清白”，重置计数
        IF v_user.last_unbind_time IS NULL OR v_user.last_unbind_time < (NOW() - INTERVAL '30 days') THEN
            v_user.unbind_count := 0;
        END IF;

        -- 判断二：检查本月换绑次数（0次、1次允许，第2次及以上拦截，即每个月只给2次机会）
        IF COALESCE(v_user.unbind_count, 0) >= 2 THEN
            RETURN json_build_object('success', false, 'msg', '本月自动换绑次数已达上限，为保障账号安全已锁定，请联系作者解绑');
        END IF;

        -- 判断三：放行！允许顶号换绑！
        -- 直接把数据库的 device_uuid 替换为当前请求的 p_uuid，换绑次数 +1，更新换绑时间
        UPDATE app_users 
        SET device_uuid = p_uuid, 
            unbind_count = COALESCE(v_user.unbind_count, 0) + 1, 
            last_unbind_time = NOW() 
        WHERE username = p_username;

        RETURN json_build_object('success', true, 'msg', '检测到新设备或数据重置，已自动为您安全换绑！', 'is_vip', v_user.is_vip);
    END IF;

END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ==========================================
-- 2. 部署 login_device_v3 (核心：强行顶号挤兑)
-- 不再检查换绑次数，直接覆写，确保“最新登录者获胜”
-- ==========================================
CREATE OR REPLACE FUNCTION login_device_v3(p_username TEXT, p_password TEXT, p_uuid TEXT)
RETURNS json AS $$
DECLARE
    v_user RECORD;
BEGIN
    SELECT * INTO v_user FROM app_users WHERE username = p_username;

    IF NOT FOUND THEN
        RETURN json_build_object('success', false, 'msg', '账号不存在');
    END IF;

    IF v_user.password != p_password THEN
        RETURN json_build_object('success', false, 'msg', '密码错误');
    END IF;

    -- 🌟 Highlander 规则：强行顶号，更新 device_uuid 为当前最新的 p_uuid
    UPDATE app_users 
    SET device_uuid = p_uuid, 
        last_login_time = NOW() 
    WHERE username = p_username;

    RETURN json_build_object(
        'success', true, 
        'msg', '登录成功', 
        'is_vip', v_user.is_vip
    );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ==========================================
-- 3. 部署 check_device_binding_v3 (核心：静默查岗)
-- ==========================================
CREATE OR REPLACE FUNCTION check_device_binding_v3(p_username TEXT, p_uuid TEXT)
RETURNS json AS $$
DECLARE
    v_db_uuid TEXT;
    v_is_vip BOOLEAN;
BEGIN
    SELECT device_uuid, is_vip INTO v_db_uuid, v_is_vip FROM app_users WHERE username = p_username;

    IF v_db_uuid = p_uuid THEN
        -- 查岗成功：当前设备仍是合法绑定的唯一设备
        RETURN json_build_object('is_bound', true, 'is_vip', v_is_vip);
    ELSE
        -- 查岗失败：账号已被其他设备顶掉
        RETURN json_build_object('is_bound', false, 'is_vip', false);
    END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
